cmake_minimum_required(VERSION 3.18)
project(px4_nmpc_cg2o LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

option(PX4_NMPC_CG2O_BUILD_PYBIND "Build the pybind11 module" ON)

# The constrained optimizer is cg2o's ISPD. PX4_NMPC_CG2O_OPTIMIZER is still accepted for compatibility
# with older build scripts and ignored.
set(PX4_NMPC_CG2O_OPTIMIZER "" CACHE STRING "ignored: the cg2o backend uses ISPD")

# -march: scripts/build_backends/cg2o.sh passes the flag of the local g2o build (must match).
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

# -----------------------------------------------------------------------------
# Strictly project-local dependency selection.
# These prefixes are passed by scripts/build_backends/cg2o.sh. CMake MUST NOT
# silently select a second g2o/cg2o installation from /usr/local.
# -----------------------------------------------------------------------------
get_filename_component(PX4_NMPC_REPO_ROOT
  "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(PX4_NMPC_G2O_PREFIX
  "${PX4_NMPC_REPO_ROOT}/install/externals/g2o" CACHE PATH "Local g2o install prefix")
set(PX4_NMPC_CG2O_PREFIX
  "${PX4_NMPC_REPO_ROOT}/install/externals/cg2o" CACHE PATH "Local cg2o install prefix")

foreach(local_prefix IN ITEMS "${PX4_NMPC_G2O_PREFIX}" "${PX4_NMPC_CG2O_PREFIX}")
  if(local_prefix MATCHES "^/usr(/|$)")
    message(FATAL_ERROR "Refusing to link g2o/cg2o from system prefix: ${local_prefix}")
  endif()
endforeach()

if(NOT EXISTS "${PX4_NMPC_G2O_PREFIX}/lib/cmake/g2o/g2oConfig.cmake")
  message(FATAL_ERROR "Project-local g2oConfig.cmake missing: ${PX4_NMPC_G2O_PREFIX}; run scripts/build_externals/g2o.sh")
endif()
if(NOT EXISTS "${PX4_NMPC_CG2O_PREFIX}/lib/cmake/cg2o/cg2oConfig.cmake")
  message(FATAL_ERROR "Project-local cg2oConfig.cmake missing: ${PX4_NMPC_CG2O_PREFIX}; run scripts/build_externals/cg2o.sh")
endif()
set(g2o_DIR "${PX4_NMPC_G2O_PREFIX}/lib/cmake/g2o" CACHE PATH "Pinned project-local g2o" FORCE)
set(cg2o_DIR "${PX4_NMPC_CG2O_PREFIX}/lib/cmake/cg2o" CACHE PATH "Pinned project-local cg2o" FORCE)
list(PREPEND CMAKE_PREFIX_PATH "${PX4_NMPC_CG2O_PREFIX}" "${PX4_NMPC_G2O_PREFIX}")
message(STATUS "Local g2o:  ${PX4_NMPC_G2O_PREFIX}")
message(STATUS "Local cg2o: ${PX4_NMPC_CG2O_PREFIX}")

# cg2o itself depends on Eigen, fmt, OpenMP and g2o. The g2o and cg2o package
# configuration files are generated by their local install steps.
find_package(Eigen3 3.3 REQUIRED)
find_package(g2o CONFIG REQUIRED PATHS "${PX4_NMPC_G2O_PREFIX}" NO_DEFAULT_PATH)
find_package(cg2o CONFIG REQUIRED PATHS "${PX4_NMPC_CG2O_PREFIX}" NO_DEFAULT_PATH)

# Complete C++ NMPC factor-graph formulation.
add_subdirectory(lib/cg2o_trajectory)

# -----------------------------------------------------------------------------
# Python binding
# -----------------------------------------------------------------------------
if(PX4_NMPC_CG2O_BUILD_PYBIND)
  # Use the active virtual environment's Python when CMake is launched from it.
  find_package(Python3 COMPONENTS Interpreter Development.Module REQUIRED)

  find_package(pybind11 CONFIG QUIET)

  # pip installs pybind11 into the virtual environment. If CMake does not find
  # its config automatically, ask the active Python interpreter for the path.
  if(NOT pybind11_FOUND)
    execute_process(
      COMMAND "${Python3_EXECUTABLE}" -m pybind11 --cmakedir
      OUTPUT_VARIABLE PYBIND11_CMAKE_DIR
      OUTPUT_STRIP_TRAILING_WHITESPACE
      ERROR_QUIET
    )

    if(PYBIND11_CMAKE_DIR)
      list(PREPEND CMAKE_PREFIX_PATH "${PYBIND11_CMAKE_DIR}")
      find_package(pybind11 CONFIG QUIET)
    endif()
  endif()

  if(NOT pybind11_FOUND)
    message(FATAL_ERROR
      "pybind11 was not found for ${Python3_EXECUTABLE}.\n"
      "Activate the project venv and run: pip install pybind11"
    )
  endif()

  add_subdirectory(bindings)

  # The extension is built standalone by scripts/build_cg2o_backend.sh.
  # solver.py discovers it in cpp/build, so cg2o stays an implementation
  # detail of px4_nmpc_controller rather than a second nested ROS package.
endif()
