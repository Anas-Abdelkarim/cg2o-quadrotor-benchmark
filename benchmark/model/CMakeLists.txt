cmake_minimum_required(VERSION 3.18)
project(challenging_mpc LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(g2o_DIR "${ROOT}/install/externals/g2o/lib/cmake/g2o")
set(cg2o_DIR "${ROOT}/install/externals/cg2o/lib/cmake/cg2o")
find_package(Eigen3 REQUIRED)
find_package(g2o CONFIG REQUIRED)
find_package(cg2o CONFIG REQUIRED)
find_package(Python3 COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" -m pybind11 --cmakedir OUTPUT_VARIABLE PYBIND_CMAKE OUTPUT_STRIP_TRAILING_WHITESPACE)
list(PREPEND CMAKE_PREFIX_PATH "${PYBIND_CMAKE}")
find_package(pybind11 CONFIG REQUIRED)
set(ENV{PKG_CONFIG_PATH} "${ROOT}/install/externals/ipopt/lib/pkgconfig:$ENV{PKG_CONFIG_PATH}")
find_package(PkgConfig REQUIRED)
pkg_check_modules(IPOPT REQUIRED IMPORTED_TARGET ipopt)
set(LEGACY "${ROOT}/src/px4_nmpc_controller/px4_nmpc_controller/px4_nmpc_cg2o/cpp/lib/cg2o_trajectory")
pybind11_add_module(_challenging NO_EXTRAS adapters.cpp "${LEGACY}/linear_solvers.cpp")
target_include_directories(_challenging PRIVATE "${LEGACY}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(_challenging PRIVATE Eigen3::Eigen cg2o::core g2o::core g2o::stuff cg2o::solver_qdldlt PkgConfig::IPOPT)
target_compile_definitions(_challenging PRIVATE PX4_NMPC_HAVE_CG2O_QDLDLT=1)
# Match the existing local Eigen/g2o ABI; all backends were built with native ISA.
target_compile_options(_challenging PRIVATE -march=native)
set_target_properties(_challenging PROPERTIES BUILD_RPATH "${ROOT}/install/externals/ipopt/lib;${ROOT}/install/externals/cg2o/lib;${ROOT}/install/externals/g2o/lib")
